# lines are skipped) so a brand-new account can find the house commands with
# CTRL-r instead of reading the wiki. {{PLACEHOLDERS}} are expanded.
#
# Order matters: the LAST lines end up nearest the top of CTRL-r results, so
# put the commands you use most in the highest-numbered file.

lsblk

systemctl --failed
journalctl -p err -b --no-pager | tail -50
ss -tulpn
sudo dmesg -T | tail -50
